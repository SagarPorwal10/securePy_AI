## 1. `securepy_ai/testgen/generator.py`

```python
"""
SecurePy AI — Phase 13: Automatic security regression test generation.

For a (finding, patch) pair, emits a standalone pytest file containing:
  - detection guard   (rule still fires on the ORIGINAL code)
  - fix guard         (rule no longer fires on the PATCHED code)
  - structure guard   (function signatures preserved)
  - CWE property test (CWE-specific static safety property)
"""
import re
from pathlib import Path

from securepy_ai.models import PatchCandidate, VulnerabilityFinding


_HEADER = '''# AUTO-GENERATED by SecurePy AI (Phase 13) — security regression tests.
# Finding: __RULE__ __CWE__ __FILE__:__LINE__
# Regenerate with: securepy-ai scan <target> --fix --gen-tests
import ast
import re
import sys

sys.path.insert(0, __ROOT__)

from securepy_ai.scanner.rules import ALL_RULES

RULE_ID = __RULE__

ORIGINAL = __ORIGINAL__

PATCHED = __PATCHED__


def _hits(code):
    try:
        tree = ast.parse(code)
    except SyntaxError:
        return None
    out = []
    for rule_class in ALL_RULES:
        rule = rule_class()
        out.extend(
            f for f in rule.scan(tree, "<generated>", code) if f.rule_id == RULE_ID
        )
    return out


def test_original_code_is_flagged():
    hits = _hits(ORIGINAL)
    assert hits is not None and len(hits) >= 1


def test_patched_code_is_clean():
    hits = _hits(PATCHED)
    assert hits is not None and len(hits) == 0


def test_patched_code_preserves_structure():
    orig = ast.parse(ORIGINAL)
    patch = ast.parse(PATCHED)
    orig_fns = {
        n.name: tuple(a.arg for a in n.args.args)
        for n in ast.walk(orig)
        if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))
    }
    patch_fns = {
        n.name: tuple(a.arg for a in n.args.args)
        for n in ast.walk(patch)
        if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))
    }
    for name, args in orig_fns.items():
        assert name in patch_fns
        assert patch_fns[name] == args

__PROPERTY__
'''


PROPERTY_TESTS = {
    "CWE-89": '''
def test_cwe89_uses_parameterized_query():
    assert ("?" in PATCHED) or ("%s" in PATCHED)
    assert not re.search(r"f[\\"']", PATCHED)
''',
    "CWE-78": '''
def test_cwe78_no_shell_execution():
    assert "shell=True" not in PATCHED
    assert "os.system" not in PATCHED
''',
    "CWE-502": '''
def test_cwe502_no_unsafe_deserialization():
    assert "pickle.loads" not in PATCHED
    assert "marshal.loads" not in PATCHED
''',
    "CWE-95": '''
def test_cwe95_no_dynamic_execution():
    assert not re.search(r"\\beval\\(", PATCHED)
    assert not re.search(r"\\bexec\\(", PATCHED)
''',
    "CWE-798": '''
def test_cwe798_no_hardcoded_secret():
    assert '= "' not in PATCHED and "= '" not in PATCHED
''',
}


class TestGenerator:
    def __init__(self, project_root=None):
        self.root = str(project_root or Path(__file__).resolve().parents[2])

    def generate(self, finding: VulnerabilityFinding, patch: PatchCandidate) -> str:
        template = _HEADER
        replacements = {
            "__ROOT__": repr(self.root),
            "__RULE__": repr(finding.rule_id),
            "__CWE__": repr(finding.cwe_id),
            "__FILE__": str(finding.file_path),
            "__LINE__": str(finding.line_number),
            "__ORIGINAL__": repr(patch.original_code),
            "__PATCHED__": repr(patch.patched_code),
            "__PROPERTY__": PROPERTY_TESTS.get(finding.cwe_id, ""),
        }
        for key, value in replacements.items():
            template = template.replace(key, value)
        return template

    def write(self, finding, patch, out_dir) -> Path:
        out = Path(out_dir)
        out.mkdir(parents=True, exist_ok=True)
        name = f"test_gen_{finding.rule_id.lower()}_{finding.line_number}.py"
        path = out / name
        path.write_text(self.generate(finding, patch), encoding="utf-8")
        return path
```

## 2. `securepy_ai/testgen/runner.py`

```python
import os
import subprocess
import sys
from pathlib import Path


def run_test_file(path, project_root=None, timeout=120) -> bool:
    """Runs one generated test file with pytest; returns True on pass."""
    root = str(project_root or Path(__file__).resolve().parents[2])

    env = os.environ.copy()
    env["PYTHONPATH"] = root + os.pathsep + env.get("PYTHONPATH", "")

    try:
        result = subprocess.run(
            [
                sys.executable, "-m", "pytest", str(path),
                "-q", "--no-header", "-p", "no:cacheprovider",
            ],
            capture_output=True,
            text=True,
            timeout=timeout,
            cwd=root,
            env=env,
        )
        return result.returncode == 0
    except Exception:
        return False
```

## 4. Updated `securepy_ai/validator/patch_validator.py`

Replace the file with this version. Key change: an optional `run_tests` mode that executes generated regression tests and folds them into the confidence score (weights rebalance to include a 0.10 test layer).

```python
import ast
import tempfile
from typing import Dict, List, Optional, Set, Tuple

from securepy_ai.models import (
    PatchCandidate,
    PatchValidation,
    ScanReport,
    ValidationCheck,
    VulnerabilityFinding,
)
from securepy_ai.scanner.rules import ALL_RULES


class PatchValidator:
    HIGH_CONFIDENCE_THRESHOLD = 0.80
    REVIEW_CONFIDENCE_THRESHOLD = 0.50

    def __init__(self, run_tests: bool = False):
        self.run_tests = run_tests

    # ------------------------------------------------------------------
    def validate_report(self, report: ScanReport) -> ScanReport:
        for finding in report.findings:
            if finding.patch is None or not finding.patch.success:
                continue
            finding.patch.validation = self.validate_finding(finding, finding.patch)
        return report

    def validate_finding(
        self, finding: VulnerabilityFinding, patch: PatchCandidate
    ) -> PatchValidation:
        checks: List[ValidationCheck] = []

        syntax_valid, syntax_msg = self._check_syntax(patch.patched_code)
        checks.append(ValidationCheck(
            "Syntax validation", "pass" if syntax_valid else "fail", syntax_msg))

        logic, logic_msg = self._check_ast_logic(patch.original_code, patch.patched_code)
        checks.append(ValidationCheck(
            "AST logic preservation", "pass" if logic else "fail", logic_msg))

        original_findings = self._scan_code(patch.original_code)
        patched_findings = self._scan_code(patch.patched_code)

        fixed, fixed_msg = self._check_vulnerability_fixed(finding, patched_findings)
        checks.append(ValidationCheck(
            "Vulnerability re-scan", "pass" if fixed else "fail", fixed_msg))

        no_new, no_new_msg = self._check_no_new_vulnerabilities(
            finding, original_findings, patched_findings)
        checks.append(ValidationCheck(
            "No new vulnerabilities", "pass" if no_new else "fail", no_new_msg))

        tests_passed = None
        if self.run_tests and patch.patched_code:
            tests_passed = self._run_generated_tests(finding, patch)

        if tests_passed is None:
            checks.append(ValidationCheck(
                "Unit tests", "skipped", "Generated regression tests not enabled."))
        else:
            checks.append(ValidationCheck(
                "Unit tests", "pass" if tests_passed else "fail",
                "Generated regression tests passed."
                if tests_passed else "Generated regression tests failed."))

        confidence = self._calculate_confidence(
            syntax_valid, logic, fixed, no_new, tests_passed)
        decision = self._make_decision(confidence, syntax_valid, fixed, no_new)
        is_valid = syntax_valid and logic and fixed and no_new

        return PatchValidation(
            is_valid=is_valid,
            syntax_valid=syntax_valid,
            ast_logic_preserved=logic,
            vulnerability_fixed=fixed,
            no_new_vulnerabilities=no_new,
            tests_passed=tests_passed,
            confidence_score=confidence,
            decision=decision,
            checks=checks,
        )

    # ------------------------------------------------------------------
    def _run_generated_tests(self, finding, patch) -> Optional[bool]:
        try:
            from securepy_ai.testgen import TestGenerator, run_test_file
            gen = TestGenerator()
            with tempfile.TemporaryDirectory() as td:
                test_path = gen.write(finding, patch, td)
                return run_test_file(test_path)
        except Exception:
            return None

    # ------------------------------------------------------------------
    def _check_syntax(self, code: str) -> Tuple[bool, str]:
        if not code.strip():
            return False, "Patch is empty."
        try:
            ast.parse(code)
            return True, "Patched code is syntactically valid."
        except SyntaxError as exc:
            return False, f"Syntax error at line {exc.lineno}: {exc.msg}"

    def _check_ast_logic(self, original: str, patched: str) -> Tuple[bool, str]:
        try:
            orig_tree = ast.parse(original)
        except SyntaxError:
            return True, "Original code could not be parsed; logic check skipped."
        try:
            patch_tree = ast.parse(patched)
        except SyntaxError:
            return False, "Patched code could not be parsed for AST comparison."

        orig_fns = self._get_function_signatures(orig_tree)
        patch_fns = self._get_function_signatures(patch_tree)
        for name, args in orig_fns.items():
            if name not in patch_fns:
                return False, f"Function '{name}' is missing after patch."
            if patch_fns[name] != args:
                return False, f"Function '{name}' signature changed."

        orig_classes = self._get_class_names(orig_tree)
        missing = orig_classes - self._get_class_names(patch_tree)
        if missing:
            return False, f"Missing classes after patch: {', '.join(sorted(missing))}"

        if not orig_fns:
            orig_targets = self._get_assignment_targets(orig_tree)
            patch_targets = self._get_assignment_targets(patch_tree)
            missing_t = orig_targets - patch_targets
            if missing_t:
                return False, ("Assignment targets not preserved: "
                               f"{', '.join(sorted(missing_t))}")

        return True, "Basic program structure appears preserved."

    def _check_vulnerability_fixed(self, finding, patched_findings) -> Tuple[bool, str]:
        ids = {f.rule_id for f in patched_findings}
        if finding.rule_id in ids:
            return False, f"{finding.rule_id} is still detected after patching."
        return True, f"{finding.rule_id} is no longer detected."

    def _check_no_new_vulnerabilities(self, finding, original_findings,
                                      patched_findings) -> Tuple[bool, str]:
        orig_ids = {f.rule_id for f in original_findings}
        patch_ids = {f.rule_id for f in patched_findings}
        new_ids = (patch_ids - orig_ids)
        new_ids.discard(finding.rule_id)
        if new_ids:
            return False, f"New vulnerabilities introduced: {', '.join(sorted(new_ids))}"
        return True, "No new vulnerability types introduced."

    def _scan_code(self, code: str) -> List[VulnerabilityFinding]:
        try:
            tree = ast.parse(code)
        except SyntaxError:
            return []
        findings = []
        for rule_class in ALL_RULES:
            rule = rule_class()
            findings.extend(rule.scan(tree, "<validator>", code))
        return findings

    def _get_function_signatures(self, tree) -> Dict[str, tuple]:
        return {
            n.name: tuple(a.arg for a in n.args.args)
            for n in ast.walk(tree)
            if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))
        }

    def _get_class_names(self, tree) -> Set[str]:
        return {n.name for n in ast.walk(tree) if isinstance(n, ast.ClassDef)}

    def _get_assignment_targets(self, tree) -> Set[str]:
        targets: Set[str] = set()
        for node in ast.walk(tree):
            if isinstance(node, ast.Assign):
                for t in node.targets:
                    self._collect_target_names(t, targets)
        return targets

    def _collect_target_names(self, target, targets):
        if isinstance(target, ast.Name):
            targets.add(target.id)
        elif isinstance(target, (ast.Tuple, ast.List)):
            for el in target.elts:
                self._collect_target_names(el, targets)

    def _calculate_confidence(self, syntax_valid, logic, fixed, no_new,
                              tests_passed) -> float:
        if not syntax_valid:
            return 0.0
        if tests_passed is None:
            score = (0.30 + (0.25 if logic else 0)
                     + (0.30 if fixed else 0) + (0.15 if no_new else 0))
        else:
            score = (0.25 + (0.20 if logic else 0)
                     + (0.30 if fixed else 0) + (0.15 if no_new else 0)
                     + (0.10 if tests_passed else 0))
        return round(min(score, 1.0), 2)

    def _make_decision(self, confidence, syntax_valid, fixed, no_new) -> str:
        if not syntax_valid or not fixed or not no_new:
            return "Reject / Manual Remediation"
        if confidence >= self.HIGH_CONFIDENCE_THRESHOLD:
            return "Auto Apply Recommended"
        if confidence >= self.REVIEW_CONFIDENCE_THRESHOLD:
            return "Developer Review Recommended"
        return "Reject / Manual Remediation"
```

from securepy_ai.models import PatchCandidate, Severity, VulnerabilityFinding
from securepy_ai.testgen import TestGenerator, run_test_file
from securepy_ai.validator.patch_validator import PatchValidator


def make_pair():
    finding = VulnerabilityFinding(
        rule_id="SEC102", vuln_type="SQL Injection", cwe_id="CWE-89",
        severity=Severity.CRITICAL, file_path="app.py", line_number=24,
        code_snippet='query = f"SELECT * FROM users WHERE id = {user_id}"',
        description="Dynamic SQL query construction detected.")
    patch = PatchCandidate(
        model="test", prompt_used="",
        original_code=('def get_user(user_id):\n'
                       '    query = f"SELECT * FROM users WHERE id = {user_id}"\n'
                       '    return db.execute(query).fetchone()\n'),
        patched_code=('def get_user(user_id):\n'
                      '    query = "SELECT * FROM users WHERE id = ?"\n'
                      '    return db.execute(query, (user_id,)).fetchone()\n'),
        raw_response="", latency_ms=0.0, success=True)
    return finding, patch


def test_generated_source_contains_guards():
    finding, patch = make_pair()
    src = TestGenerator().generate(finding, patch)
    assert "test_original_code_is_flagged" in src
    assert "test_patched_code_is_clean" in src
    assert "test_cwe89_uses_parameterized_query" in src


def test_generated_tests_pass(tmp_path):
    finding, patch = make_pair()
    path = TestGenerator().write(finding, patch, tmp_path)
    assert run_test_file(path) is True


def test_validator_run_tests_layer():
    finding, patch = make_pair()
    validator = PatchValidator(run_tests=True)
    validation = validator.validate_finding(finding, patch)
    assert validation.tests_passed is True
    assert validation.confidence_score == 1.0
